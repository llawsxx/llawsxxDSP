cmake_minimum_required(VERSION 3.20)
project(llawsxxDSPVST2 C CXX)
set(VST2_SDK "${CMAKE_CURRENT_SOURCE_DIR}/third_party/VST_SDK_2.4" CACHE PATH "VST2 SDK root")
if(NOT EXISTS "${VST2_SDK}/pluginterfaces/vst2.x/aeffect.h")
  set(VST2_SDK "${CMAKE_CURRENT_SOURCE_DIR}/third_party/vst2sdk")
endif()
if(NOT EXISTS "${VST2_SDK}/pluginterfaces/vst2.x/aeffect.h")
  message(FATAL_ERROR "VST2 SDK not found. Set -DVST2_SDK=<path>.")
endif()
add_library(llawsxxDSP SHARED simpledsp_vst2.cpp simpledsp_editor.cpp convolution_reverb.c "${VST2_SDK}/public.sdk/source/vst2.x/audioeffect.cpp" "${VST2_SDK}/public.sdk/source/vst2.x/audioeffectx.cpp")
target_include_directories(llawsxxDSP PRIVATE "${VST2_SDK}" "${VST2_SDK}/pluginterfaces/vst2.x" "${VST2_SDK}/public.sdk/source/vst2.x")
target_compile_definitions(llawsxxDSP PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
target_compile_features(llawsxxDSP PRIVATE cxx_std_17)
if(WIN32)
  target_link_libraries(llawsxxDSP PRIVATE user32 gdi32)
endif()
if(MSVC)
  target_compile_options(llawsxxDSP PRIVATE /W4 /wd4100 /wd4458 /wd4996)
else()
  target_compile_options(llawsxxDSP PRIVATE -Wall -Wextra -ffast-math)
endif()
set_target_properties(llawsxxDSP PROPERTIES PREFIX "" OUTPUT_NAME "llawsxxDSP")

include(CTest)
if(BUILD_TESTING AND WIN32)
  add_executable(vst2_smoke tests/vst2_smoke.cpp)
  target_include_directories(vst2_smoke PRIVATE "${VST2_SDK}/pluginterfaces/vst2.x")
  target_compile_definitions(vst2_smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(vst2_smoke PRIVATE imm32)
  add_test(NAME vst2_smoke COMMAND vst2_smoke "$<TARGET_FILE:llawsxxDSP>")
endif()
